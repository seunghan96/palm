# Every arm the paper reports, one per line:
#
#     <arm>  <flags passed to palm/adapt.py>
#
# The arm is the directory under results/big and the name every table refers
# to. Two placeholders are filled per checkpoint by run/02_adapt.sh:
#
#     {PS}   the backbone's own setting from configs/settings.txt,
#            written out as --rank R --steps S --lr LR
#     {DOM}  --fit-window LO HI with HI the checkpoint's cutoff and LO the same
#            distance before it as the decision date is after it, floored at
#            2014.0 where FNSPID begins. Only vintages cut from 2015.0 on have
#            a year of FNSPID before them, the others are skipped.
#
# The decision date is 2021.0 for every arm here. The validation arms move it
# to 2018.0 and are generated by `bash run/02_adapt.sh val`, not listed.

# the checkpoint as released, scored through the same path as every other arm
base_test     --no-train

# PALM at the reported setting
sel_ps        {PS}

# same budget, text the vintage's era already covers: domain without recency
dom_ps        {PS} {DOM}

# the two bounds: the words scrambled, and text read past the decision date
shuf_ps       {PS} --shuffle-text
oracle_ps     {PS} --fit-to 2024.0

# alternatives that also train on the same text
full_ft       --rank 2 --steps 600 --lr 1e-5 --full
sft_r2        --rank 2 --steps 600 --lr 1e-4 --sft

# same-firm retrieval at scoring time, without and with PALM
base_hist     --no-train --ctx hist --k 5
sel_ps_hist   {PS} --ctx hist --k 5

# robustness
mask_ps       {PS} --mask-pad
seed1_ps      {PS} --seed 1
seed2_ps      {PS} --seed 2
seed3_ps      {PS} --seed 3
